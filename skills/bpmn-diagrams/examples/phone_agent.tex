\begin{document}
\begin{tikzpicture}[dp, card/.style={box, text width=1.85cm, minimum height=0.9cm}]
\exlanes
\node[bpmessage] (s) at (2.55,1.45) {};
\node[bpq, anchor=north] at ([yshift=-1pt]s.south) {call\\arrives};
\node[card, text width=1.5cm] (c1) at (3.75,1.45) {Open a bounded job for the call};
\node[card] (m1) at (5.9,2.9) {Interpret the request; propose a tool call};
\node[bpgw] (g2) at (5.9,1.45) {};
\node[bpq, anchor=north west, align=left] at ([xshift=1pt]g2.south east) {caller on and\\turns left?};
\node[bpend] (e2) at (5.9,0.55) {};
\node[bpq, anchor=east, align=right] at ([xshift=-2pt]e2.west) {ended\\by code};
\node[card, text width=1.5cm] (ck) at (8.5,1.45) {Check hours, slot lock, request key};
\node[bpgw] (g1) at (9.9,1.45) {};
\node[bpq, anchor=north] at ([yshift=-1pt]g1.south) {pass?};
\node[card] (c3) at (11.6,1.45) {Run the calendar or CRM tool; return facts and a verdict};
\node[card] (m2) at (11.6,2.9) {Read the result; choose the next step};
\node[bpopen] (gm) at (13.35,2.9) {};
\node[bpq, anchor=north east] at ([xshift=-1pt]gm.south west) {next\\step?};
\node[card, text width=1.1cm, minimum height=0.55cm] (a1) at (15.05,3.1) {Answer the caller};
\node[bpend] (e1) at (15.05,2.42) {};
\node[bpq, anchor=west] at ([xshift=1pt]e1.east) {call\\ended};
\node[card, text width=1.25cm] (h1) at (13.35,-0.45) {Staff member takes the transfer};
\node[bpend] (e3) at (14.75,-0.45) {};
\node[bpq, anchor=south] at ([yshift=1pt]e3.north) {transferred};
\draw[flow] (s) -- (c1);
\draw[flow] (c1.north) |- (m1.west);
\draw[flow] (m1) -- (g2);
\draw[flow] (g2) -- node[lab, above] {yes} (ck);
\draw[flow] (g2) -- node[lab, left] {no} (e2);
\draw[flow] (ck) -- (g1);
\draw[flow] (g1) -- node[lab, above] {yes} (c3);
\draw[flow] (g1.north) |- node[lab, pos=0.75, above] {no: rejected} (m1.east);
\draw[flow] (c3) -- (m2);
\draw[flow] (m2) -- (gm);
\draw[flow] (gm.east) -- node[lab, above] {answer} (gm.east -| a1.west);
\draw[flow] (gm.south east) |- node[lab, pos=0.75, below] {end} (e1.west);
\draw[flow] (a1.north) -- ++(0,0.55) -| node[lab, pos=0.12, below] {next turn} ([xshift=-0.3cm]m1.north);
\draw[flow] (gm.south) -- node[lab, pos=0.75, right] {transfer} (h1.north);
\draw[flow] (h1) -- (e3);
\draw[flow] (gm.north) -- ++(0,0.3) -| node[lab, pos=0.31, below] {book: next tool call} ([xshift=0.3cm]m1.north);
\exlegend{legopen}{\node[bpmessage, minimum size=0.3cm] at (14.75,0) {}; \node[lab, anchor=west] at (14.95,0) {message};}
